// AeroORM client initialization
import { AeroClient } from '@aero/orm';

const db = new AeroClient({
  url: process.env.DATABASE_URL,
  logLevel: 'perf'
});

// Type-safe ultra-fast query
const users = await db.users.findMany({
  where: { status: 'ACTIVE' },
  select: { id: true, email: true },
  limit: 100000
});

const report = await db.orders.findMany({
  with: {
    customer: { select: { name: true } },
    items: {
      with: { product: true }
    }
  },
  where: (orders, { gte }) => gte(orders.total, 1500),
  orderBy: (orders, { desc }) => desc(orders.createdAt)
});
